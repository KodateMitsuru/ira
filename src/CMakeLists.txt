cmake_minimum_required(VERSION 3.22)
project(ira_runtime LANGUAGES C CXX)
get_filename_component(IRA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(IRA_SHELL "" CACHE FILEPATH "Built Android session shell for signal tests")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_library(ira-platform-options INTERFACE)
target_compile_definitions(ira-platform-options INTERFACE _GNU_SOURCE _FILE_OFFSET_BITS=64)
if(NOT ANDROID)
  find_package(Threads REQUIRED)
  file(GLOB _ndk_sysroots "${IRA_NDK}/toolchains/llvm/prebuilt/*/sysroot/usr/include")
  list(GET _ndk_sysroots 0 IRA_NDK_HEADERS)
  if(NOT EXISTS "${IRA_NDK_HEADERS}/android/log.h")
    message(FATAL_ERROR "NDK public headers are required for host declarations")
  endif()
  target_compile_options(ira-platform-options INTERFACE -idirafter "${IRA_NDK_HEADERS}")
  target_link_libraries(ira-platform-options INTERFACE Threads::Threads)
endif()
add_subdirectory("${IRA_ROOT}/vendor/boringssl" "${CMAKE_BINARY_DIR}/crypto")
add_subdirectory(base)
add_subdirectory(auth)
add_subdirectory(adb)
if(ANDROID)
  add_subdirectory(mksh)
  add_subdirectory(zygisk)
else()
  enable_testing()
  add_subdirectory("${IRA_ROOT}/tests" "${CMAKE_BINARY_DIR}/tests")
endif()
